local function GenerateKey()
    local chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    local key = ""
    for i = 1, 4 do
        for j = 1, 4 do
            key = key .. chars:sub(math.random(1, #chars), math.random(1, #chars))
        end
        if i < 4 then key = key .. "-" end
    end
    return key
end

local function HashKey(key)
    local hash = 0
    for i = 1, #key do
        hash = (hash * 31 + string.byte(key, i)) % 2^31
    end
    return tostring(hash)
end

local function ValidateKey(input, expectedHash)
    return HashKey(string.upper(input:gsub("%s+", ""))) == expectedHash
end

-- Safe UI creation wrapper (bypasses most executors' anti)
local function CreateInstance(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props) do
        pcall(function() inst[k] = v end)
    end
    return inst
end

local function MakeDraggable(frame, dragObj)
    dragObj = dragObj or frame
    local dragging, dragInput, dragStart, startPos
    dragObj.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    dragObj.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function Tween(obj, props, duration, style)
    style = style or Enum.EasingStyle.Quad
    local tween = TweenService:Create(obj, TweenInfo.new(duration or 0.3, style, Enum.EasingDirection.Out), props)
    tween:Play()
    return tween
end

    function UI:SelectTab(name)
        for _, tab in pairs(UI.Tabs) do
            local isSelected = tab.Data.Name == name
            tab.Button.BackgroundTransparency = isSelected and 0 or 0.5
            tab.Button.TextColor3 = isSelected and Draconis.Theme.Accent or Draconis.Theme.TextDim
            tab.Container.Visible = isSelected
            if isSelected then
                Tween(tab.Button, {BackgroundColor3 = Draconis.Theme.Accent:Lerp(Draconis.Theme.Element, 0.7)}, 0.2)
            else
                Tween(tab.Button, {BackgroundColor3 = Draconis.Theme.Element}, 0.2)
            end
        end
        UI.CurrentTab = name
    end

    function UI:AddSlider(text, min, max, default, suffix, callback, sectionData)
        sectionData = sectionData or UI.CurrentSection
        if not sectionData then return end
        
        local value = default or min
        suffix = suffix or ""
        
        local sliderFrame = CreateInstance("Frame", {
            Name = text,
            BackgroundColor3 = Draconis.Theme.Element,
            BorderSizePixel = 0,
            Size = UDim2.new(1, 0, 0, 50),
            Parent = sectionData.Content,
        })
        
        local sliderCorner = CreateInstance("UICorner", {
            CornerRadius = UDim.new(0, 6),
            Parent = sliderFrame,
        })
        
        local sliderLabel = CreateInstance("TextLabel", {
            Name = "Label",
            BackgroundTransparency = 1,
            Font = Enum.Font.Gotham,
            Text = text,
            TextColor3 = Draconis.Theme.Text,
            TextSize = 13,
            Position = UDim2.new(0, 12, 0, 8),
            Size = UDim2.new(0.5, -12, 0, 20),
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = sliderFrame,
        })
        
        local valueLabel = CreateInstance("TextLabel", {
            Name = "Value",
            BackgroundTransparency = 1,
            Font = Enum.Font.GothamBold,
            Text = tostring(value) .. suffix,
            TextColor3 = Draconis.Theme.Accent,
            TextSize = 13,
            Position = UDim2.new(0.5, 0, 0, 8),
            Size = UDim2.new(0.5, -12, 0, 20),
            TextXAlignment = Enum.TextXAlignment.Right,
            Parent = sliderFrame,
        })
        
        -- Slider background
        local sliderBg = CreateInstance("Frame", {
            Name = "SliderBg",
            BackgroundColor3 = Draconis.Theme.Background,
            BorderSizePixel = 0,
            Position = UDim2.new(0, 12, 0, 34),
            Size = UDim2.new(1, -24, 0, 8),
            Parent = sliderFrame,
        })
        
        local sliderBgCorner = CreateInstance("UICorner", {
            CornerRadius = UDim.new(0, 4),
            Parent = sliderBg,
        })
        
        -- Slider fill
        local sliderFill = CreateInstance("Frame", {
            Name = "SliderFill",
            BackgroundColor3 = Draconis.Theme.Accent,
            BorderSizePixel = 0,
            Size = UDim2.new((value - min) / (max - min), 0, 1, 0),
            Parent = sliderBg,
        })
        
        local sliderFillCorner = CreateInstance("UICorner", {
            CornerRadius = UDim.new(0, 4),
            Parent = sliderFill,
        })
        
        -- Draggable slider
        local sliderInput = CreateInstance("TextButton", {
            Name = "SliderInput",
            BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 1, 0),
            Text = "",
            Parent = sliderBg,
        })
        
        local dragging = false
        sliderInput.MouseButton1Down:Connect(function()
            dragging = true
        end)
        
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
        
        local function UpdateSlider(inputPos)
            local pos = sliderBg.AbsolutePosition
            local size = sliderBg.AbsoluteSize.X
            local relX = math.clamp(inputPos.X - pos.X, 0, size)
            local newValue = min + (relX / size) * (max - min)
            newValue = math.floor(newValue * 100) / 100
            value = newValue
            sliderFill.Size = UDim2.new(relX / size, 0, 1, 0)
            valueLabel.Text = tostring(value) .. suffix
            pcall(callback, value)
        end
        
        sliderInput.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                UpdateSlider(input)
            end
        end)
        
        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                UpdateSlider(input)
            end
        end)
        
        return {
            Set = function(v)
                value = math.clamp(v, min, max)
                sliderFill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
                valueLabel.Text = tostring(value) .. suffix
            end,
            Get = function() return value end,
        }
    end

    -- ============================================================
    -- WATERMARK
    -- ============================================================
    
    function UI:AddWatermark(text)
        local watermark = CreateInstance("TextLabel", {
            Name = "Watermark",
            BackgroundColor3 = Draconis.Theme.Surface,
            BackgroundTransparency = 0.2,
            BorderSizePixel = 0,
            Font = Enum.Font.GothamBold,
            Text = text or "Draconis UI",
            TextColor3 = Draconis.Theme.Accent,
            TextSize = 13,
            Position = UDim2.new(0, 10, 0, 10),
            Size = UDim2.new(0, 0, 0, 28),
            AutomaticSize = Enum.AutomaticSize.X,
            Parent = screenGui,
        })
        
        local watermarkCorner = CreateInstance("UICorner", {
            CornerRadius = UDim.new(0, 6),
            Parent = watermark,
        })
        
        local padding = CreateInstance("UIPadding", {
            PaddingLeft = UDim.new(0, 12),
            PaddingRight = UDim.new(0, 12),
            Parent = watermark,
        })
        
        -- FPS counter
        local fpsLabel = CreateInstance("TextLabel", {
            Name = "FPS",
            BackgroundTransparency = 1,
            Font = Enum.Font.Gotham,
            Text = "60 FPS",
            TextColor3 = Draconis.Theme.TextDim,
            TextSize = 11,
            Position = UDim2.new(0, 0, 0, 28),
            Size = UDim2.new(0, 60, 0, 20),
            Parent = watermark,
        })
        
        local lastTime = tick()
        local frames = 0
        coroutine.wrap(function()
            while true do
                task.wait()
                frames = frames + 1
                local elapsed = tick() - lastTime
                if elapsed >= 1 then
                    local fps = math.floor(frames / elapsed)
                    fpsLabel.Text = fps .. " FPS"
                    frames = 0
                    lastTime = tick()
                end
            end
        end)()
        
        return watermark
    end

    UI.ScreenGui = screenGui

    return UI
